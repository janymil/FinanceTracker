 Wrap Your App with <QueryClientProvider>
If not already done, in your main.jsx or _app.tsx:

tsx
Copy
Edit
import { QueryClient, QueryClientProvider } from 'react-query'

const queryClient = new QueryClient()

<QueryClientProvider client={queryClient}>
  <App />
</QueryClientProvider>
📦 2. Fetch Transactions with useQuery
Wherever you display transactions (Dashboard, Ingresos, etc.):

tsx
Copy
Edit
const { data: transactions, isLoading } = useQuery(
  ['transactions', userId], // key should be unique per user
  () => fetchTransactions(userId), // fetch function
  {
    staleTime: 0, // always consider data stale
    refetchOnWindowFocus: false,
    refetchOnReconnect: true,
  }
)
✍️ 3. Use useMutation for Editing/Creating/Deleting
In the component where you create/edit/delete a transaction:

tsx
Copy
Edit
const queryClient = useQueryClient()

const mutation = useMutation(saveTransaction, {
  onSuccess: () => {
    queryClient.invalidateQueries(['transactions', userId])
  }
})
👉 Example:
tsx
Copy
Edit
const handleSave = async () => {
  await mutation.mutateAsync({ amount: 100, description: "Test" })
}
This tells React Query: “Hey, the transactions may be outdated — refetch now.”

🧠 4. (Optional) Show Immediate Optimistic Updates (Advanced)
If you want even faster feedback (e.g. showing the new transaction before the API call finishes):

tsx
Copy
Edit
const mutation = useMutation(saveTransaction, {
  onMutate: async (newData) => {
    await queryClient.cancelQueries(['transactions', userId])
    const previous = queryClient.getQueryData(['transactions', userId])
    queryClient.setQueryData(['transactions', userId], (old) => [...old, newData])
    return { previous }
  },
  onError: (err, newData, context) => {
    queryClient.setQueryData(['transactions', userId], context.previous)
  },
  onSettled: () => {
    queryClient.invalidateQueries(['transactions', userId])
  }
})
