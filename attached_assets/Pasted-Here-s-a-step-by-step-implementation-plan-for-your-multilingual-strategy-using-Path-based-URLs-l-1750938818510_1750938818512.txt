✅ 2. Setup i18next
i18n/i18n.js
js
Copy
Edit
import i18n from 'i18next';
import { initReactI18next } from 'react-i18next';
import HttpApi from 'i18next-http-backend';
import LanguageDetector from './languageDetector';

i18n
  .use(HttpApi)
  .use(LanguageDetector)
  .use(initReactI18next)
  .init({
    fallbackLng: 'en-us',
    supportedLngs: ['en-us', 'es-pe', 'sk-sk'],
    backend: {
      loadPath: '/locales/{{lng}}/translation.json',
    },
    interpolation: {
      escapeValue: false,
    },
    detection: {
      order: ['path', 'cookie', 'navigator'],
      caches: ['cookie'],
    }
  });

export default i18n;
i18n/languageDetector.js (Custom to detect from path)
js
Copy
Edit
const LanguageDetector = {
  type: 'languageDetector',
  async: true,
  detect: (cb) => {
    const pathLang = window.location.pathname.split('/')[1]; // e.g., "es-pe"
    if (['es-pe', 'en-us', 'sk-sk'].includes(pathLang)) {
      cb(pathLang);
    } else {
      cb('en-us');
    }
  },
  init: () => {},
  cacheUserLanguage: () => {}
};

export default LanguageDetector;

✅ 5. Language Redirect on First Visit
src/utils/redirectToLocale.js

export async function redirectToLocale() {
  if (window.location.pathname === "/") {
    const res = await fetch("https://ipapi.co/json"); // or ipinfo.io
    const { country_code, languages } = await res.json();
    const lang = (languages?.split(',')[0] || 'en').toLowerCase();
    const path = `${lang}-${country_code.toLowerCase()}`;
    window.location.href = `/${path}`;
  }
}
Call this in index.jsx before rendering the app.

✅ 7. AI Prompt Translation via DeepL + Caching
translatePrompt.js
js
Copy
Edit
const cache = new Map();

export async function translatePrompt(prompt, targetLang) {
  const key = `${targetLang}-${prompt}`;
  if (cache.has(key)) return cache.get(key);

  const res = await fetch("https://api-free.deepl.com/v2/translate", {
    method: "POST",
    headers: { "Content-Type": "application/x-www-form-urlencoded" },
    body: `auth_key=YOUR_API_KEY&text=${encodeURIComponent(prompt)}&target_lang=${targetLang.toUpperCase()}`
  });
  const data = await res.json();
  const translated = data.translations[0].text;
  cache.set(key, translated);
  return translated;
}
Example in AI Function:

import { translatePrompt } from './translatePrompt';

async function generateAIResponse(userInput, lang) {
  const prompt = `Explain the monthly spending for this user: ${userInput}`;
  const localizedPrompt = await translatePrompt(prompt, lang);
  
  const response = await openai.chat.completions.create({
    model: 'gpt-4',
    messages: [{ role: 'system', content: localizedPrompt }],
  });

  return response.choices[0].message.content;
}
